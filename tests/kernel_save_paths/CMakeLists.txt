cmake_minimum_required(VERSION 3.20)
project(kernel_save_paths_test C)
set(CMAKE_C_STANDARD 11)
if(NOT WIN32)
    message(FATAL_ERROR "This temporary-files regression requires the Win32 path backend")
endif()

target_include_directories(kernel_save_paths_test PRIVATE ../../src ../../src/kernel)
target_link_libraries(kernel_save_paths_test PRIVATE shell32)
enable_testing()
add_test(NAME kernel_save_paths_test COMMAND kernel_save_paths_test)
