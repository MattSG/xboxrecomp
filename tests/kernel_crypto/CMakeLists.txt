cmake_minimum_required(VERSION 3.20)

set(CMAKE_C_STANDARD 11)
add_executable(kernel_crypto_test test_main.c ../../src/kernel/kernel_crypto.c)
target_include_directories(kernel_crypto_test PRIVATE ../../src ../../src/kernel)

endif()
enable_testing()
