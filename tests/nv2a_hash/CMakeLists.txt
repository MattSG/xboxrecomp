cmake_minimum_required(VERSION 3.20)
project(nv2a_hash C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()
add_executable(nv2a_hash test_hash.c)
target_include_directories(nv2a_hash PRIVATE ../../src/video)
if(NOT WIN32)
    # The Win32 primitives (VirtualAlloc/VirtualProtect, QPC) the test uses.
    add_subdirectory(../../src/platform platform)
    target_link_libraries(nv2a_hash PRIVATE platform m)
endif()
foreach(threshold 1024 2048 4096)
    add_library(hash_${threshold} OBJECT test_dispatch.c)
    target_include_directories(hash_${threshold} PRIVATE ../../src/video)
    target_compile_definitions(hash_${threshold} PRIVATE
        NV_HASH_NATIVE_MIN=${threshold}u NV_HASH_TEST_NAME=threshold_${threshold})
    target_sources(nv2a_hash PRIVATE $<TARGET_OBJECTS:hash_${threshold}>)
endforeach()
foreach(case exact pages)
    add_test(NAME nv2a_hash_${case} COMMAND nv2a_hash ${case})
endforeach()
# The CPU feature dispatch is the MSVC x64 build of nv2a_texture_hash.h only.
if(MSVC)
    add_test(NAME nv2a_hash_features COMMAND nv2a_hash features)
endif()
