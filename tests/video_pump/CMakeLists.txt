cmake_minimum_required(VERSION 3.20)
project(video_pump_test C)
enable_testing()
if(WIN32)
    add_executable(video_pump_test test_main.c)
    target_compile_features(video_pump_test PRIVATE c_std_11)
    target_include_directories(video_pump_test PRIVATE ../../src)
    target_link_libraries(video_pump_test PRIVATE user32)
    add_test(NAME video_pump_callbacks COMMAND video_pump_test)
else()
    add_executable(video_pump_stub_test test_stub.c
        ../../src/video/video_pump.c ../../src/video/video_player.c)
    target_include_directories(video_pump_stub_test PRIVATE ../../src)
    add_test(NAME video_pump_stubs COMMAND video_pump_stub_test)
endif()
