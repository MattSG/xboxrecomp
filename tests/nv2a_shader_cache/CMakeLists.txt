cmake_minimum_required(VERSION 3.20)
project(nv2a_shader_cache C)
set(CMAKE_C_STANDARD 11)
enable_testing()
# The SHA-256 and cache key run everywhere; the D3DCompile file cache is Windows-only.
add_executable(nv2a_shader_cache_key test_key.c)
target_include_directories(nv2a_shader_cache_key PRIVATE ../../src ../../src/video)
add_test(NAME nv2a_shader_cache_key COMMAND nv2a_shader_cache_key)
if(NOT WIN32)
    message(STATUS "nv2a_shader_cache: D3DCompile file cache test, Windows only")
    return()
endif()
add_executable(nv2a_shader_cache test_main.c)
target_include_directories(nv2a_shader_cache PRIVATE ../../src/video)
target_link_libraries(nv2a_shader_cache PRIVATE d3dcompiler)
add_test(NAME nv2a_shader_cache COMMAND nv2a_shader_cache)
