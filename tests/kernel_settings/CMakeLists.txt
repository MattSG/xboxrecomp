cmake_minimum_required(VERSION 3.20)
project(kernel_settings_test C)
set(CMAKE_C_STANDARD 11)
add_subdirectory(../../src/platform platform)
add_executable(kernel_settings_test test_main.c
    ../../src/kernel/kernel_xbox.c ../../src/kernel/kernel_path.c)
target_include_directories(kernel_settings_test PRIVATE ../../src/kernel)
target_link_libraries(kernel_settings_test PRIVATE platform)
if(WIN32)
    target_link_libraries(kernel_settings_test PRIVATE shell32)
else()
    target_link_libraries(kernel_settings_test PRIVATE m)
endif()
enable_testing()
add_test(NAME kernel_settings COMMAND kernel_settings_test)
set_tests_properties(kernel_settings PROPERTIES TIMEOUT 30)
