cmake_minimum_required(VERSION 3.20)
project(net_regression LANGUAGES C)
if(WIN32)
    set(NET_TEST_LIBS ws2_32 iphlpapi)
else()
    # The Win32 primitives net_backend.c is built on (events, SRW locks, threads).
    add_subdirectory(../../src/platform platform)
    set(NET_TEST_LIBS platform m)
endif()
add_executable(net-test net_test.c)
target_include_directories(net-test PRIVATE ../../src)
target_link_libraries(net-test PRIVATE ${NET_TEST_LIBS})
enable_testing()
add_test(NAME net COMMAND net-test)
add_executable(nvnet_ring_test nvnet_ring_test.c)
target_include_directories(nvnet_ring_test PRIVATE ../../src)
target_link_libraries(nvnet_ring_test PRIVATE ${NET_TEST_LIBS})
if(WIN32)
    target_link_libraries(nvnet_ring_test PRIVATE winmm)
endif()
target_compile_definitions(nvnet_ring_test PRIVATE _CRT_SECURE_NO_WARNINGS)
add_test(NAME nvnet_ring COMMAND nvnet_ring_test)
