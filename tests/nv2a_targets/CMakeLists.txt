cmake_minimum_required(VERSION 3.20)
project(nv2a_targets C)
enable_testing()
if(WIN32)
    include(../../cmake/Nv2aHlsl.cmake)   # the renderer's embedded HLSL
    add_executable(nv2a_targets test_targets.c)
    target_include_directories(nv2a_targets PRIVATE ../../src)
    target_link_libraries(nv2a_targets PRIVATE d3d11 dxgi dxguid d3dcompiler)
    foreach(samples 1 2 4 8)
        add_test(NAME nv2a_targets_${samples}x COMMAND nv2a_targets ${samples})
    endforeach()
    xbr_embed_nv2a_hlsl(nv2a_targets)
endif()

# The SDL3 GPU renderer's targets and guest textures, on every platform its backend builds on.
include(../../cmake/Nv2aSdlgpu.cmake)
add_subdirectory(../../src/platform ${CMAKE_CURRENT_BINARY_DIR}/platform)
add_executable(nv2a_targets_sdlgpu test_targets_sdlgpu.c ../../src/video/fb_window_sdl.c ../../src/input/keyboard.c
    ../../src/input/keymap_spec.c ../../src/input/input_map.c)
xbr_nv2a_sdlgpu_target(nv2a_targets_sdlgpu)
target_link_libraries(nv2a_targets_sdlgpu PRIVATE platform)
if(NOT MSVC)
    target_compile_options(nv2a_targets_sdlgpu PRIVATE -fwrapv -fno-strict-aliasing)
endif()
foreach(case sampled_upload lru texture_rewrite palette_rewrite writeback_replacement writeback_retained swizzled_sampled compressed_palette y16_aliases common)
    add_test(NAME nv2a_targets_sdlgpu_${case} COMMAND nv2a_targets_sdlgpu ${case})
endforeach()
