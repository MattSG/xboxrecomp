cmake_minimum_required(VERSION 3.20)
project(kernel_audio_setting_test C)
set(CMAKE_C_STANDARD 11)
# Standalone, or part of a combined tree that already has the runtime.
if(NOT TARGET xboxrecomp)
    add_subdirectory(../.. runtime)
endif()
add_executable(kernel_audio_setting_test test_main.c)
target_include_directories(kernel_audio_setting_test PRIVATE ../../src/kernel)
target_link_libraries(kernel_audio_setting_test PRIVATE xboxrecomp)
xbox_copy_runtime_dlls(kernel_audio_setting_test)
enable_testing()
add_test(NAME kernel_audio_setting_test COMMAND kernel_audio_setting_test)
