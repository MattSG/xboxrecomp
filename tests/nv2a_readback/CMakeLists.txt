cmake_minimum_required(VERSION 3.20)
project(nv2a_readback C)
enable_testing()
if(WIN32)
    include(../../cmake/Nv2aHlsl.cmake)   # the renderer's embedded HLSL
    add_executable(nv2a_readback test_main.c)
    target_include_directories(nv2a_readback PRIVATE ../../src ../../src/video)
    target_link_libraries(nv2a_readback PRIVATE d3d11 dxgi dxguid d3dcompiler)
    foreach(samples 1 2 4 8)
        add_test(NAME nv2a_readback_batch_${samples}x COMMAND nv2a_readback batch ${samples})
    endforeach()
    foreach(case query_create_retry query_failure map_retry drop_resize down_rtv_retry trace_off trace_on)
        add_test(NAME nv2a_readback_${case} COMMAND nv2a_readback ${case})
    endforeach()
    xbr_embed_nv2a_hlsl(nv2a_readback)
endif()

# The SDL3 GPU renderer's batched readback, on every platform its backend builds on.
include(../../cmake/Nv2aSdlgpu.cmake)
add_subdirectory(../../src/platform ${CMAKE_CURRENT_BINARY_DIR}/platform)
add_executable(nv2a_readback_sdlgpu test_main_sdlgpu.c ../../src/video/fb_window_sdl.c)
xbr_nv2a_sdlgpu_target(nv2a_readback_sdlgpu)
target_link_libraries(nv2a_readback_sdlgpu PRIVATE platform)
if(NOT MSVC)
    target_compile_options(nv2a_readback_sdlgpu PRIVATE -fwrapv -fno-strict-aliasing)
endif()
foreach(case batch fence_create_retry map_retry drop_resize down_retry common)
    add_test(NAME nv2a_readback_sdlgpu_${case} COMMAND nv2a_readback_sdlgpu ${case})
endforeach()
