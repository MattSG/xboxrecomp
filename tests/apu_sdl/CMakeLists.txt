cmake_minimum_required(VERSION 3.20)
project(apu_sdl_test C)
include(../../cmake/Sdl3.cmake)
add_executable(apu_sdl_test test_main.c)
target_link_libraries(apu_sdl_test PRIVATE SDL3::SDL3)
if(MSVC)
