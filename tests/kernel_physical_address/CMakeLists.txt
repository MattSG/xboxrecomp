cmake_minimum_required(VERSION 3.20)
project(kernel_physical_address_test C)
if(NOT WIN32)
    message(FATAL_ERROR "This synthetic-memory regression uses Windows VirtualAlloc")
endif()
set(CMAKE_C_STANDARD 11)
# Standalone, or part of a combined tree that already has the runtime.
if(NOT TARGET xboxrecomp)
    add_subdirectory(../.. runtime)
endif()
add_executable(kernel_physical_address_test test_main.c)
target_include_directories(kernel_physical_address_test PRIVATE ../../src/kernel)
target_link_libraries(kernel_physical_address_test PRIVATE xboxrecomp)
xbox_copy_runtime_dlls(kernel_physical_address_test)
enable_testing()
add_test(NAME kernel_physical_address_test COMMAND kernel_physical_address_test)
