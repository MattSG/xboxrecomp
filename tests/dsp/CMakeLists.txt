cmake_minimum_required(VERSION 3.20)
project(dsp_regression C)
set(DSP_XEMU_REFERENCE_DIR "" CACHE PATH "Pinned xemu source tree for independent CPU/DMA checks")
enable_testing()
add_library(dsp_runtime STATIC ../../src/apu/dsp/interp/dsp_cpu.c
  ../../src/apu/dsp/dsp_dma.c ../../src/apu/dsp/dsp.c)
target_include_directories(dsp_runtime PUBLIC ../../src/apu/dsp)
target_compile_features(dsp_runtime PRIVATE c_std_11)
add_executable(dsp_cpu_test dsp_cpu_test.c)
target_link_libraries(dsp_cpu_test PRIVATE dsp_runtime)
target_compile_features(dsp_cpu_test PRIVATE c_std_11)
target_include_directories(dsp_cpu_test PRIVATE ../../src/apu/dsp)
if(MSVC)
  target_compile_definitions(dsp_runtime PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_definitions(dsp_cpu_test PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_test(NAME dsp_cpu COMMAND dsp_cpu_test)
add_executable(dsp_disasm_test dsp_disasm_test.c)
target_include_directories(dsp_disasm_test PRIVATE ../../src/apu/dsp)
target_compile_definitions(dsp_disasm_test PRIVATE _CRT_SECURE_NO_WARNINGS)
add_test(NAME dsp_disasm COMMAND dsp_disasm_test)
add_executable(dsp_frame_replay dsp_frame_replay.c)
target_link_libraries(dsp_frame_replay PRIVATE dsp_runtime)
target_compile_definitions(dsp_frame_replay PRIVATE _CRT_SECURE_NO_WARNINGS)
if(DSP_XEMU_REFERENCE_DIR)
  execute_process(COMMAND git -C "${DSP_XEMU_REFERENCE_DIR}" rev-parse HEAD
    OUTPUT_VARIABLE xemu_commit OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE xemu_git_result)
  execute_process(COMMAND git -C "${DSP_XEMU_REFERENCE_DIR}" diff HEAD --quiet -- hw/xbox/mcpx/apu/dsp
    RESULT_VARIABLE xemu_dirty_result)
  if(NOT xemu_git_result EQUAL 0 OR NOT xemu_dirty_result EQUAL 0 OR
     NOT xemu_commit STREQUAL "478b4f496102379c7eaa7f3ec10e714a703c4300")
    message(FATAL_ERROR "DSP reference must be the unchanged pinned xemu DSP sources")
  endif()
  set(xemu_dsp "${DSP_XEMU_REFERENCE_DIR}/hw/xbox/mcpx/apu/dsp")
  add_executable(dsp_xemu_cpu_test dsp_cpu_test.c "${xemu_dsp}/interp/dsp_cpu.c")
  target_include_directories(dsp_xemu_cpu_test PRIVATE reference ../../src/apu/dsp "${xemu_dsp}")
  target_compile_features(dsp_xemu_cpu_test PRIVATE c_std_11)
  target_compile_definitions(dsp_xemu_cpu_test PRIVATE DSP_REFERENCE_CPU_ONLY _CRT_SECURE_NO_WARNINGS)
  add_test(NAME dsp_xemu_cpu COMMAND dsp_xemu_cpu_test)
  add_executable(dsp_xemu_frame_replay dsp_frame_replay.c ../../src/apu/dsp/dsp.c
    ../../src/apu/dsp/dsp_dma.c "${xemu_dsp}/interp/dsp_cpu.c")
  target_include_directories(dsp_xemu_frame_replay PRIVATE reference ../../src/apu/dsp "${xemu_dsp}")
  target_compile_definitions(dsp_xemu_frame_replay PRIVATE _CRT_SECURE_NO_WARNINGS)
  add_executable(dsp_dma_compare dsp_dma_compare.c)
  target_link_libraries(dsp_dma_compare PRIVATE dsp_runtime)
  add_executable(dsp_xemu_dma_compare dsp_dma_compare.c "${xemu_dsp}/dsp_dma.c")
  target_include_directories(dsp_xemu_dma_compare PRIVATE reference "${xemu_dsp}" ../../src/apu/dsp)
  target_compile_definitions(dsp_xemu_dma_compare PRIVATE DSP_REFERENCE_DMA _CRT_SECURE_NO_WARNINGS)
  if(MSVC)
    target_compile_definitions(dsp_dma_compare PRIVATE _CRT_SECURE_NO_WARNINGS)
  endif()
  add_test(NAME dsp_dma_reference COMMAND "${CMAKE_COMMAND}"
    "-Dport=$<TARGET_FILE:dsp_dma_compare>" "-Dreference=$<TARGET_FILE:dsp_xemu_dma_compare>"
    "-Doutput=${CMAKE_CURRENT_BINARY_DIR}/dma-results" -P "${CMAKE_CURRENT_SOURCE_DIR}/compare_dma.cmake")
endif()
if(WIN32)
  add_executable(dsp_apu_test dsp_apu_test.c apu_dsp_under_test.c)
  target_include_directories(dsp_apu_test PRIVATE ../../src ../../src/apu)
  target_compile_definitions(dsp_apu_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(dsp_apu_test PRIVATE dsp_runtime)
  add_test(NAME dsp_apu COMMAND dsp_apu_test)
  add_test(NAME dsp_frame_capture COMMAND "${CMAKE_COMMAND}" -E env
    "RECOMP_DSP_FRAME_DUMP=${CMAKE_CURRENT_BINARY_DIR}/test-frame" "$<TARGET_FILE:dsp_apu_test>")
  add_test(NAME dsp_capture_reset COMMAND "${CMAKE_COMMAND}" -E env
    "RECOMP_DSP_FRAME_DUMP=${CMAKE_CURRENT_BINARY_DIR}/reset-frame"
    "DSP_TEST_CAPTURE_RESET=1" "$<TARGET_FILE:dsp_apu_test>")
  add_test(NAME dsp_wait_capture COMMAND "${CMAKE_COMMAND}" -E env
    "RECOMP_DSP_FRAME_DUMP=${CMAKE_CURRENT_BINARY_DIR}/wait-frame"
    "DSP_TEST_CAPTURE_WAIT=1" "$<TARGET_FILE:dsp_apu_test>")
  set_tests_properties(dsp_wait_capture PROPERTIES FIXTURES_SETUP captured_wait_frame TIMEOUT 10)
  add_test(NAME dsp_wait_replay COMMAND dsp_frame_replay "${CMAKE_CURRENT_BINARY_DIR}/wait-frame.gp.frame.bin")
  set_tests_properties(dsp_wait_replay PROPERTIES FIXTURES_REQUIRED captured_wait_frame TIMEOUT 10)
  add_test(NAME dsp_legacy_capture COMMAND "${CMAKE_COMMAND}" -E env
    "RECOMP_DSP_FRAME_DUMP=${CMAKE_CURRENT_BINARY_DIR}/legacy-frame"
    "DSP_TEST_CAPTURE_LEGACY=1" "$<TARGET_FILE:dsp_apu_test>")
  set_tests_properties(dsp_legacy_capture PROPERTIES FIXTURES_SETUP legacy_frame TIMEOUT 10)
  add_test(NAME dsp_legacy_rejected COMMAND dsp_frame_replay "${CMAKE_CURRENT_BINARY_DIR}/legacy-frame.gp.frame.bin")
  set_tests_properties(dsp_legacy_rejected PROPERTIES FIXTURES_REQUIRED legacy_frame WILL_FAIL TRUE TIMEOUT 10)
  set_tests_properties(dsp_frame_capture PROPERTIES FIXTURES_SETUP captured_frame)
  set_tests_properties(dsp_apu dsp_frame_capture PROPERTIES TIMEOUT 10)
  foreach(processor IN ITEMS gp ep)
    add_test(NAME dsp_replay_${processor} COMMAND dsp_frame_replay "${CMAKE_CURRENT_BINARY_DIR}/test-frame.${processor}.frame.bin")
    set_tests_properties(dsp_replay_${processor} PROPERTIES FIXTURES_REQUIRED captured_frame)
    if(DSP_XEMU_REFERENCE_DIR)
      add_test(NAME dsp_xemu_replay_${processor} COMMAND dsp_xemu_frame_replay "${CMAKE_CURRENT_BINARY_DIR}/test-frame.${processor}.frame.bin")
      set_tests_properties(dsp_xemu_replay_${processor} PROPERTIES FIXTURES_REQUIRED captured_frame)
    endif()
  endforeach()
endif()

set_tests_properties(dsp_clipped_replay PROPERTIES FIXTURES_REQUIRED clipped_frame)
