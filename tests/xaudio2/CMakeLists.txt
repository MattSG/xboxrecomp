cmake_minimum_required(VERSION 3.20)
project(xaudio2_test C)
if(NOT WIN32)
    # POSIX backend: the SDL3 audio stream, on SDL's dummy audio driver.
    list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../cmake")
    include(Sdl3)
    enable_testing()
    add_executable(xaudio2_sdl_test sdl_test.c)
    target_link_libraries(xaudio2_sdl_test PRIVATE SDL3::SDL3)
    add_test(NAME xaudio2_sdl_test COMMAND xaudio2_sdl_test)
    set_tests_properties(xaudio2_sdl_test PROPERTIES TIMEOUT 60
        ENVIRONMENT "SDL_AUDIO_DRIVER=dummy")
    return()
endif()
set(CMAKE_C_STANDARD 11)
add_executable(xaudio2_test test_main.c)
enable_testing()
add_test(NAME xaudio2_test COMMAND xaudio2_test)

# Extract the actual function rather than copying its wait logic into a test.
set(monitor_source "${CMAKE_CURRENT_SOURCE_DIR}/../../src/apu/apu_core.c")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${monitor_source}")
file(READ "${monitor_source}" monitor_code)
string(FIND "${monitor_code}" "void mcpx_apu_monitor_frame(" monitor_begin)
string(FIND "${monitor_code}" "static void throttle(" monitor_end)
if(monitor_begin LESS 0 OR monitor_end LESS monitor_begin)
  message(FATAL_ERROR "Cannot locate production monitor body")
endif()
math(EXPR monitor_length "${monitor_end} - ${monitor_begin}")
string(SUBSTRING "${monitor_code}" ${monitor_begin} ${monitor_length} monitor_body)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/monitor_under_test.inc" "${monitor_body}")
add_executable(monitor_test monitor_test.c)
target_include_directories(monitor_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME monitor_test COMMAND monitor_test)
set_tests_properties(monitor_test PROPERTIES TIMEOUT 5)
