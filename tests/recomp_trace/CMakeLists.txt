cmake_minimum_required(VERSION 3.20)
project(trace_reads C)
add_subdirectory(../../src/platform platform)
add_executable(trace_reads test_main.c)
target_link_libraries(trace_reads PRIVATE platform)
enable_testing()
add_test(NAME trace_reads COMMAND ${CMAKE_COMMAND} -E env --unset=RECOMP_TRACE_PROFILE --unset=RECOMP_TRACE_ARGS --unset=RECOMP_TRACE_DEREF RECOMP_TRACE_BUDGET=100 $<TARGET_FILE:trace_reads>)
add_test(NAME trace_args COMMAND ${CMAKE_COMMAND} -E env --unset=RECOMP_TRACE_PROFILE RECOMP_TRACE_BUDGET=100 RECOMP_TRACE_ARGS=2 RECOMP_TRACE_DEREF=1 $<TARGET_FILE:trace_reads>)
set_tests_properties(trace_reads trace_args PROPERTIES PASS_REGULAR_EXPRESSION "GUEST.*valid")
