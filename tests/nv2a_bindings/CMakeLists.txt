cmake_minimum_required(VERSION 3.20)
project(nv2a_bindings C)
enable_testing()
if(WIN32)
    include(../../cmake/Nv2aHlsl.cmake)   # the renderer's embedded HLSL
    add_executable(nv2a_bindings test_bindings.c)
    target_include_directories(nv2a_bindings PRIVATE ../../src ../../src/video)
    target_link_libraries(nv2a_bindings PRIVATE d3d11 dxgi dxguid d3dcompiler)
    foreach(case pixels fixed_updates failures resize init_failure ps_bindings ps_failure ps_reset overlay_upload host_draw)
        add_test(NAME nv2a_bindings_${case} COMMAND nv2a_bindings ${case})
    endforeach()
    xbr_embed_nv2a_hlsl(nv2a_bindings)
endif()

# The SDL3 GPU renderer's draw bindings, on every platform its backend builds on.
include(../../cmake/Nv2aSdlgpu.cmake)
add_subdirectory(../../src/platform ${CMAKE_CURRENT_BINARY_DIR}/platform)
add_executable(nv2a_bindings_sdlgpu test_bindings_sdlgpu.c ../../src/video/fb_window_sdl.c)
xbr_nv2a_sdlgpu_target(nv2a_bindings_sdlgpu)
target_link_libraries(nv2a_bindings_sdlgpu PRIVATE platform)
if(NOT MSVC)
    target_compile_options(nv2a_bindings_sdlgpu PRIVATE -fwrapv -fno-strict-aliasing)
endif()
foreach(case pixels failures resize init_failure ps_bindings ps_reset overlay_upload host_draw common)
    add_test(NAME nv2a_bindings_sdlgpu_${case} COMMAND nv2a_bindings_sdlgpu ${case})
endforeach()
